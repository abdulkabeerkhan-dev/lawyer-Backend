def verify_reviewed_tier(canonical_id: str, content_hash: str) -> bool:
    """
    Order 1.14 - Item 6: Cryptographic public-key signature verification for 'reviewed' tier.
    The authorized advocate public key is committed in data/reviews/advocate_public_key.pem.
    The corresponding private key DOES NOT EXIST on this machine and is never generated here.
    An agent/model cannot sign reviews; any signature made with a throwaway key fails verification.
    """
    review_file = os.path.join(REVIEWS_DIR, f"{canonical_id}.review.json")
    pub_key_file = os.path.join(REVIEWS_DIR, "advocate_public_key.pem")
    
    if not os.path.exists(review_file) or not os.path.exists(pub_key_file):
        return False
    try:
        with open(review_file, "r", encoding="utf-8") as rf:
            data = json.load(rf)
        if str(data.get("canonical_id")).upper() != str(canonical_id).upper():
            return False
        if str(data.get("record_content_hash")) != str(content_hash):
            return False
        if data.get("status") != "approved":
            return False
        
        sig_hex = data.get("cryptographic_signature", "")
        if not sig_hex:
            return False
        sig_bytes = bytes.fromhex(sig_hex)

        from cryptography.hazmat.primitives import hashes, serialization
        from cryptography.hazmat.primitives.asymmetric import padding
        
        with open(pub_key_file, "rb") as pf:
            public_key = serialization.load_pem_public_key(pf.read())
        
        # Payload verified is canonical_id + content_hash
        payload = f"{canonical_id}:{content_hash}".encode("utf-8")
        
        public_key.verify(
            sig_bytes,
            payload,
            padding.PKCS1v15(),
            hashes.SHA256()
        )
        return True
    except Exception:
        return False

-- 3. BATCHED BACKFILL PROCEDURE WITH TSVECTOR 1MB LIMIT TRUNCATION
-- ----------------------------------------------------------------------------
-- PostgreSQL tsvector has a hard 1MB (1,048,575 bytes) internal limit.
-- Landmark apex judgments exceeding ~100 pages will crash to_tsvector with:
--   "ERROR: string is too long for tsvector"
-- We safely truncate full_text to the first 500,000 characters using `substring(coalesce(fj.full_text, '') from 1 for 500000)`.
CREATE OR REPLACE PROCEDURE backfill_full_judgments_fts_batched(batch_size INT DEFAULT 5000)
LANGUAGE plpgsql
AS $$
DECLARE
    rows_updated INT := 0;
    total_updated INT := 0;
BEGIN
    LOOP
        WITH rows_to_update AS (
            SELECT id 
            FROM public.full_judgments 
            WHERE fts_tokens IS NULL 
            LIMIT batch_size
            FOR UPDATE SKIP LOCKED
        )
        UPDATE public.full_judgments fj
        SET fts_tokens = (
            setweight(to_tsvector('english', coalesce(fj.case_title, '')), 'A') ||
            setweight(to_tsvector('english', coalesce(fj.neutral_citation, '')), 'A') ||
            setweight(to_tsvector('english', coalesce(fj.court_name, '')), 'B') ||
            setweight(to_tsvector('english', substring(coalesce(fj.full_text, '') from 1 for 500000)), 'C')
        )
        FROM rows_to_update
        WHERE fj.id = rows_to_update.id;

        GET DIAGNOSTICS rows_updated = ROW_COUNT;
        total_updated := total_updated + rows_updated;
        
        RAISE NOTICE 'Batched FTS Backfill: Updated % rows (Total: %)', rows_updated, total_updated;
        
        COMMIT;

        IF rows_updated = 0 THEN
            EXIT;
        END IF;
    END LOOP;

    RAISE NOTICE 'Batched FTS Backfill COMPLETE. Total rows indexed: %', total_updated;
END;
$$;

CREATE OR REPLACE FUNCTION search_judgments_fts(
    query_text TEXT,
    court_filter TEXT DEFAULT NULL,
    match_limit INT DEFAULT 30
)
RETURNS TABLE (
    id UUID,
    case_id TEXT,
    neutral_citation TEXT,
    case_title TEXT,
    court_name TEXT,
    decision_date DATE,
    rank_score REAL,
    snippet TEXT
)
LANGUAGE sql
STABLE
AS $$
    SELECT 
        fj.id,
        fj.case_id,
        fj.neutral_citation,
        fj.case_title,
        fj.court_name,
        fj.decision_date,
        ts_rank_cd(fj.fts_tokens, websearch_to_tsquery('english', query_text), 32 /* rank normalization */) AS rank_score,
        ts_headline('english', coalesce(fj.full_text, ''), websearch_to_tsquery('english', query_text),

    FROM public.full_judgments fj
    WHERE 
        fj.fts_tokens @@ websearch_to_tsquery('english', query_text)
        AND (court_filter IS NULL OR fj.court_name ILIKE ('%' || court_filter || '%'))
    ORDER BY rank_score DESC
    LIMIT match_limit;
$$;
